# This is a self-running graphics demo: it clears to a random color, scatters
# random semigraphics blocks, and loops forever. A good moving-image test for
# the video output.

10 CLS RND(8)
20 FOR I=1 TO 60
30 PRINT @ RND(511), CHR$(128+RND(127));
40 NEXT I
50 GOTO 10
RUN
